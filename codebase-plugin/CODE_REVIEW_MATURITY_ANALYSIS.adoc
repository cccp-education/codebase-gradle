// Cucumber Task Factory (DRY Pattern)

data class CucumberTaskSpec(
    val taskName: String,
    val description: String,
    val runnerClass: String? = null,
    val timeoutMinutes: Long = 15,
)

fun isCI() = System.getenv("CI") == "true"

fun Project.registerCucumberTask(spec: CucumberTaskSpec): TaskProvider<Test> {
    val maxMemoryMB = Runtime.getRuntime().maxMemory() / (1024 * 1024)
    val adaptiveHeap = when {
        isCI() -> "2g"
        maxMemoryMB < 8192 -> "512m"
        else -> "1g"
    }

    return tasks.register<Test>(spec.taskName) {
        description = spec.description
        group = "verification"
        
        testClassesDirs = sourceSets.test.get().output.classesDirs
        classpath = configurations.testRuntimeClasspath.get() +
            sourceSets.test.get().output +
            sourceSets.main.get().output +
            files(tasks.jar.get().archiveFile)

        dependsOn(tasks.classes)
        useJUnitPlatform { excludeEngines("junit-jupiter") }
        systemProperty("cucumber.junit-platform.naming-strategy", "long")
        maxHeapSize = adaptiveHeap
        maxParallelForks = 1
        forkEvery = 100
        timeout.set(Duration.ofMinutes(spec.timeoutMinutes))

        testLogging {
            events("passed", "skipped", "failed")
            showStandardStreams = true
            exceptionFormat = FULL
        }
        outputs.upToDateWhen { false }

        spec.runnerClass?.let { filter { includeTestsMatching(it) } }
    }
}

**Location**: `codebase-plugin/build.gradle.kts` (lines 19–22, 92–95)

==== Heap Calculation Problem

**Scenario**: Developer with 8GB RAM runs all 24 Cucumber tasks in parallel

....
24 tasks × 1GB per task = 24GB required
Developer machine has: 8GB
Result: OutOfMemoryError, system thrashing, build hangs
....

**Current Configuration Analysis**:

[cols="2,2,2"]
|===
| Setting | Current | Problem

| `maxHeapSize` | 1g | Excessive for local (OOM on 8GB), inadequate for CI
| `maxParallelForks` | 1 | Forces sequential (slow), doesn't scale
| `forkEvery` | 1 | Overhead: spawn JVM per test
| `useSerialGC` | ✓ | Outdated for modern JVMs
| `TieredStopAtLevel` | 1 | No JIT optimization (slow long tests)
|===

==== Recommended Solution: Adaptive Tuning

[source,kotlin]
----
// ============================================================================
// Adaptive JVM Configuration
// ============================================================================

fun isCI() = System.getenv("CI") == "true"

fun adaptiveHeap() = when {
    isCI() -> "2g"                      // CI: more resources
    Runtime.getRuntime().maxMemory() < 8_000_000_000L -> "512m"  // <8GB: small
    else -> "1g"                        // Default dev: 1GB
}

val jvmGcArgs = when {
    isCI() -> {
        // CI: optimize for throughput (heavy workloads)
        listOf(
            "-XX:+UseG1GC",
            "-XX:MaxGCPauseMillis=200",
            "-XX:+ParallelRefProcEnabled"
        )
    }
    else -> {
        // Local: optimize for startup speed
        listOf(
            "-XX:+UseSerialGC",
            "-XX:TieredStopAtLevel=4"  // Allow some JIT
        )
    }
}

val metaspaceArgs = listOf(
    "-XX:MaxMetaspaceSize=512m",       // More headroom for Kotlin
    "-XX:CompressedClassSpaceSize=256m"
)

// ============================================================================
// Global Test Configuration
// ============================================================================

tasks.withType<Test>().configureEach {
    // Only ignore failures for non-critical tasks in local dev
    ignoreFailures = !isCI()  // ⚠️ Currently set to 'true' unconditionally
    
    useJUnitPlatform()
    
    // Enable dynamic agent loading for LLM tools (koog-agents, langchain4j)
    jvmArgs("-XX:+EnableDynamicAgentLoading")
    
    // Apply adaptive GC + metaspace configuration
    jvmArgs(jvmGcArgs + metaspaceArgs)
}

// ============================================================================
// Adaptive Cucumber Task Configuration
// ============================================================================

val cucumberTasks = cucumberTaskSpecs.map { spec ->
    tasks.register<Test>(spec.taskName) {
        // ... base config (see CRITICAL-1 refactor above)
        
        // Adaptive heap based on environment
        maxHeapSize = adaptiveHeap()
        
        // Scale parallelization based on CPU cores
        maxParallelForks = when {
            isCI() -> Runtime.getRuntime().availableProcessors()
            else -> 1  // Local: serial to avoid conflicts
        }
        
        // Reuse JVMs to reduce spawn overhead
        forkEvery = 100  // Instead of 1
        
        // Increase timeouts for safety (pgvector startup + LLM calls are slow)
        timeout.set(Duration.ofMinutes(spec.timeoutMinutes))
    }
}
----

==== Benefits

[cols="1,2"]
|===
| Benefit | Details

| **Reduces OOM Risk** | Local heap: 1GB → 512MB on <8GB systems (50% savings)
| **Enables Parallelization** | When RAM available, runs multiple tasks concurrently
| **Reduces JVM Overhead** | forkEvery=1 → forkEvery=100 (reuse JVM across tests)
| **Better GC Strategy** | G1GC in CI (predictable pauses), SerialGC locally (startup)
| **Safer CI** | Escalates heap to 2GB for heavy CI workloads
|===

**Effort**: ~20 minutes

3. **No caching**: Gradle cache not utilized (even though `org.gradle.caching=true` configured)

==== Recommended Multi-Stage CI/CD

[source,yaml]
----
# ============================================================================
# .github/workflows/ci.yml - Production-Ready CI/CD
# ============================================================================

name: CI/CD Pipeline

on:
  push:
    branches: [main, develop]
    tags: ['v*']
  pull_request:
    branches: [main, develop]

jobs:
  # ============================================================================
  # Stage 1: Fast Tests (PR Gate)
  # ============================================================================
  fast-tests:
    name: Fast Tests (PR Gate)
    runs-on: ubuntu-latest
    timeout-minutes: 10

    steps:
      - uses: actions/checkout@v4
      - name: Set up JDK 24
        uses: actions/setup-java@v4
        with:
          java-version: '24'
          distribution: 'temurin'
          cache: gradle

      - name: Run Fast Tests (JUnit5 + Quick Cucumber)
        run: ./gradlew testFast -x build --no-build-cache
        env:
          CI: "true"

  # ============================================================================
  # Stage 2: Full Tests (Main Branch Only)
  # ============================================================================
  full-tests:
    name: Full Test Suite
    runs-on: ubuntu-latest
    timeout-minutes: 45
    if: github.ref == 'refs/heads/main'
    needs: [fast-tests]

    services:
      postgres:
        image: pgvector/pgvector:pg17
        env:
          POSTGRES_DB: codebase_test
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432

    steps:
      - uses: actions/checkout@v4
      - name: Set up JDK 24
        uses: actions/setup-java@v4
        with:
          java-version: '24'
          distribution: 'temurin'
          cache: gradle

      - name: Run All Tests
        run: ./gradlew testAll --no-build-cache
        env:
          CI: "true"
          DATABASE_URL: jdbc:postgresql://localhost:5432/codebase_test

      - name: Upload Test Reports
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: test-reports
          path: codebase-plugin/build/reports/

      - name: Upload Coverage
        uses: codecov/codecov-action@v3
        with:
          files: ./codebase-plugin/build/reports/kover/report.xml

  # ============================================================================
  # Stage 3: Security Scanning
  # ============================================================================
  security:
    name: Security & Dependencies
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - uses: actions/checkout@v4
      - name: Set up JDK 24
        uses: actions/setup-java@v4
        with:
          java-version: '24'
          distribution: 'temurin'
          cache: gradle

      - name: Validate Dependencies (CVE Check)
        run: ./gradlew validateDependencies

      - name: Run Dependency Check
        run: ./gradlew dependencyCheckAnalyze || true

  # ============================================================================
  # Stage 4: Build & Publish (Tags Only)
  # ============================================================================
  publish:
    name: Publish to Maven Central
    runs-on: ubuntu-latest
    timeout-minutes: 20
    if: startsWith(github.ref, 'refs/tags/v')
    needs: [fast-tests, full-tests, security]

    steps:
      - uses: actions/checkout@v4
      - name: Set up JDK 24
        uses: actions/setup-java@v4
        with:
          java-version: '24'
          distribution: 'temurin'

      - name: Import GPG Key
        uses: crazy-max/ghaction-import-gpg@v6
        with:
          gpg_private_key: ${{ secrets.GPG_PRIVATE_KEY }}
          passphrase: ${{ secrets.GPG_PASSPHRASE }}

      - name: Publish to Maven Central
        run: ./gradlew publishToMavenCentral
        env:

== Part 6: Performance Analysis

=== Local Vibecoding Loop Bottlenecks

**Current Timeline**:

[cols="1,2,1"]
|===
| Phase | Duration | Notes

| Clone repo | 2 min | One-time
| gradlew publishToMavenLocal | 45s | One-time
| Start PostgreSQL | 30s | One-time
| Load Ollama/Gemini | 5s | One-time
| **Total Setup** | **~3-4 min** | One-time
| Edit Kotlin | 10s | Per cycle
| gradlew classes (incremental) | 8-15s | Per cycle
| OpenCode LLM planning | 2-4s | Per cycle
| Gradle task execution | 3-5s | Per cycle
| Validation checks | 2-3s | Per cycle
| **Per-Cycle Total** | **25-37s** | Iterative (good)
|===

**Full Vibecoding (first run)**:

[cols="1,2"]
|===
| Phase | Time

| Compile + dependencies | 45s
| publishToMavenLocal | 20s
| Load plugin into project | 15s
| pgvector indexing | 2-5 min ← **BOTTLENECK**
| Execute first task | 3-5s
| **Total** | **~3-5.5 min**
|===

**Root Cause**: pgvector indexing dominates. Embedding + insert for large codebases.

==== Recommendations

1. **Cache pgvector data** between sessions
2. **Batch embeddings** (chunk files, parallelize)
3. **Index incrementally** (only new/changed files)
4. **Pre-compute embeddings** offline
